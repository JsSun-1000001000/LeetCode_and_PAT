## [25. K 个一组翻转链表 - 力扣（LeetCode）](https://leetcode.cn/problems/reverse-nodes-in-k-group/description/)

## 代码
```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    //左闭右开
    ListNode* reverseList(ListNode* start, ListNode* end) {
        ListNode* temp; //临时存下一个节点
        ListNode* pre = nullptr;    //新头
        ListNode* cur = start;  //当前要翻转节点
        //左闭右开 翻转到end前一个为止
        while (cur != end) {
            temp = cur->next;   //保存下一个
            cur->next = pre;    //翻转当前节点
            pre = cur;          //pre移动到当前
            cur = temp;         //cur移动到下一个
        }
        return pre; //返回翻转后新的头节点
    }
    ListNode* reverseKGroup(ListNode* head, int k) {
        //dummy node
        ListNode* dummy = new ListNode(0);
        dummy->next = head;

        ListNode* pre = dummy;  //pre记录每组翻转前的前驱节点
        ListNode* start = head; //起始
        ListNode* end = head;   //结束节点的下一个

        while (true) {
            int count = 0;  //定位
            while (end != nullptr && count < k) {
                end = end->next;
                count++;
            }
            //不足k直接退出
            if (count < k) {
                break;
            }

            //翻转 得到新头
            ListNode* reverseHead = reverseList(start, end);
            //前驱指向翻转后新头
            pre->next = reverseHead;
            //原start变为尾节点指向end
            start->next = end;

            //更新指针 准备下一组
            pre = start;
            start = end;

        }
        //返回新的头节点 dummy的next
        ListNode* newHead = dummy->next;
        delete dummy;
        return newHead;
    }
};
```
